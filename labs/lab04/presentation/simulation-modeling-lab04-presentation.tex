\section{Введение}\label{ux432ux432ux435ux434ux435ux43dux438ux435}

\begin{frame}[fragile]{Цель и задачи}
\protect\phantomsection\label{ux446ux435ux43bux44c-ux438-ux437ux430ux434ux430ux447ux438}
\textbf{Цель:} реализовать агентную модель SIR на Julia
(\texttt{Agents.jl}) и исследовать влияние параметров, пространственной
структуры, миграции и карантинных мер.

\textbf{Задачи:}

\begin{itemize}[<+->]
\tightlist
\item
  реализовать агентную модель SIR с пространственной структурой из
  городов;
\item
  реализовать миграцию, передачу инфекции, выздоровление и смертность;
\item
  исследовать порог эпидемии, гетерогенность городов, миграцию,
  карантин;
\item
  выполнить оптимизацию параметров;
\item
  получить Julia-код, Jupyter Notebook и Quarto-документацию.
\end{itemize}
\end{frame}

\section{Теоретические
сведения}\label{ux442ux435ux43eux440ux435ux442ux438ux447ux435ux441ux43aux438ux435-ux441ux432ux435ux434ux435ux43dux438ux44f}

\begin{frame}[fragile]{Классическая и агентная модель SIR}
\protect\phantomsection\label{ux43aux43bux430ux441ux441ux438ux447ux435ux441ux43aux430ux44f-ux438-ux430ux433ux435ux43dux442ux43dux430ux44f-ux43cux43eux434ux435ux43bux44c-sir}
Классическая модель:

\[
\frac{dS}{dt}=-\beta\frac{SI}{N}, \quad
\frac{dI}{dt}=\beta\frac{SI}{N}-\gamma I, \quad
\frac{dR}{dt}=\gamma I
\]

В агентной реализации каждый человек --- отдельный агент \texttt{Person}
(\texttt{GraphAgent}) со статусом \texttt{:S}, \texttt{:I}, \texttt{:R}
и счётчиком дней болезни.
\end{frame}

\begin{frame}[fragile]{Пространственная структура и параметры}
\protect\phantomsection\label{ux43fux440ux43eux441ux442ux440ux430ux43dux441ux442ux432ux435ux43dux43dux430ux44f-ux441ux442ux440ux443ux43aux442ux443ux440ux430-ux438-ux43fux430ux440ux430ux43cux435ux442ux440ux44b}
\begin{itemize}[<+->]
\tightlist
\item
  три города на полном графе (каждый связан с двумя другими);
\item
  \texttt{Ns\ =\ {[}1000,\ 1000,\ 1000{]}} --- население городов;
\item
  \texttt{β\_und}, \texttt{β\_det} --- вероятность передачи
  (невыявленные/выявленные);
\item
  \texttt{infection\_period\ =\ 14}, \texttt{detection\_time\ =\ 7},
  \texttt{death\_rate\ =\ 0.02};
\item
  \texttt{reinfection\_probability\ =\ 0.1}, \texttt{seed\ =\ 42},
  \texttt{n\_steps\ =\ 100};
\item
  миграция задаётся матрицей \texttt{migration\_rates}.
\end{itemize}
\end{frame}

\section{Реализация
модели}\label{ux440ux435ux430ux43bux438ux437ux430ux446ux438ux44f-ux43cux43eux434ux435ux43bux438}

\begin{frame}[fragile]{Логика модели}
\protect\phantomsection\label{ux43bux43eux433ux438ux43aux430-ux43cux43eux434ux435ux43bux438}
\begin{itemize}[<+->]
\tightlist
\item
  \textbf{Инициализация:} \texttt{initialize\_sir} создаёт города,
  заполняет агентами, один инфицированный стартует в третьем городе.
\item
  \textbf{Шаг агента:} миграция → передача инфекции → увеличение дней
  болезни → выздоровление/смерть.
\item
  \textbf{Передача инфекции:} только между агентами в одном городе,
  интенсивность зависит от статуса выявления.
\item
  \textbf{Миграция:} вероятности перехода задаются матрицей
  \texttt{migration\_rates}.
\end{itemize}
\end{frame}
